#include <ESP8266WiFi.h>
#include <espnow.h>

// LEDs
#define L1 D1
#define L2 D2
#define L3 D3
#define L4 D4

typedef struct {
  int line;
} Message;

Message msg;

// ✅ Receive Callback
void OnDataRecv(uint8_t *mac, uint8_t *incomingData, uint8_t len) {
  memcpy(&msg, incomingData, sizeof(msg));

  Serial.print("Received: ");
  Serial.println(msg.line);

  // Sab OFF
  digitalWrite(L1, LOW);
  digitalWrite(L2, LOW);
  digitalWrite(L3, LOW);
  digitalWrite(L4, LOW);

  // Ek ON
  if (msg.line == 1) digitalWrite(L1, HIGH);
  else if (msg.line == 2) digitalWrite(L2, HIGH);
  else if (msg.line == 3) digitalWrite(L3, HIGH);
  else if (msg.line == 4) digitalWrite(L4, HIGH);
}

void setup() {
  Serial.begin(115200);

  Serial.println("\n=== RECEIVER START ===");

  pinMode(L1, OUTPUT);
  pinMode(L2, OUTPUT);
  pinMode(L3, OUTPUT);
  pinMode(L4, OUTPUT);

  WiFi.mode(WIFI_STA);

  Serial.print("Receiver MAC: ");
  Serial.println(WiFi.macAddress());

  if (esp_now_init() != 0) {
    Serial.println("ESP-NOW Failed");
    return;
  }

  esp_now_set_self_role(ESP_NOW_ROLE_SLAVE);
  esp_now_register_recv_cb(OnDataRecv);

  Serial.println("Receiver Ready");
}

void loop() {
}

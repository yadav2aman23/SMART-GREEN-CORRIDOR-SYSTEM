#define L1 2
#define L2 3
#define L3 4
#define L4 5

void setup() {
  Serial.begin(9600);

  pinMode(L1, OUTPUT);
  pinMode(L2, OUTPUT);
  pinMode(L3, OUTPUT);
  pinMode(L4, OUTPUT);
}

void loop() {
  if (Serial.available()) {
    int data = Serial.parseInt();

    digitalWrite(L1, LOW);
    digitalWrite(L2, LOW);
    digitalWrite(L3, LOW);
    digitalWrite(L4, LOW);

    if (data == 1) digitalWrite(L1, HIGH);
    else if (data == 2) digitalWrite(L2, HIGH);
    else if (data == 3) digitalWrite(L3, HIGH);
    else if (data == 4) digitalWrite(L4, HIGH);
  }
}
